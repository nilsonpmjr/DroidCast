cmake_minimum_required(VERSION 3.21)
project(DroidCastPrototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_library(compact-engine ../src/engine.cpp ../src/engine.h)
target_link_libraries(compact-engine PUBLIC Qt6::Widgets)
target_include_directories(compact-engine PUBLIC ../src)
add_library(prototype-ui window.cpp window.h livewindow.cpp livewindow.h)
target_link_libraries(prototype-ui PUBLIC compact-engine)
target_include_directories(prototype-ui PUBLIC .)
qt_add_executable(droidcast-prototype main.cpp)
target_link_libraries(droidcast-prototype PRIVATE prototype-ui)
set_target_properties(droidcast-prototype PROPERTIES MACOSX_BUNDLE TRUE
  MACOSX_BUNDLE_GUI_IDENTIFIER "org.droidcast.prototype"
  MACOSX_BUNDLE_BUNDLE_NAME "DroidCast Prototype" WIN32_EXECUTABLE TRUE)
set(DROIDCAST_PROTOTYPE_RUNTIME_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../build-droidcast/runtime"
    CACHE PATH "Development-only runtime staging source for the compact app")
if(EXISTS "${DROIDCAST_PROTOTYPE_RUNTIME_SOURCE}/scrcpy-server")
  add_custom_command(TARGET droidcast-prototype POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${DROIDCAST_PROTOTYPE_RUNTIME_SOURCE}"
      "$<TARGET_FILE_DIR:droidcast-prototype>/runtime")
else()
  message(WARNING "No staged runtime: demo works but live mode requires a complete fork runtime.")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(compact-monitor-probe monitor_probe.cpp)
  target_link_libraries(compact-monitor-probe PRIVATE compact-engine)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(prototype-tests test_window.cpp)
  target_link_libraries(prototype-tests PRIVATE prototype-ui Qt6::Test)
  add_test(NAME prototype-tests COMMAND prototype-tests)
  set_tests_properties(prototype-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(compact-fake-tool ../tests/fake_tool.cpp)
  target_link_libraries(compact-fake-tool PRIVATE Qt6::Gui)
  add_executable(compact-live-tests test_livewindow.cpp)
  target_link_libraries(compact-live-tests PRIVATE prototype-ui Qt6::Test)
  add_dependencies(compact-live-tests compact-fake-tool)
  add_test(NAME compact-live-tests COMMAND compact-live-tests)
  set_tests_properties(compact-live-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
